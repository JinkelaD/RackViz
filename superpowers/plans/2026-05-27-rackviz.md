# RackViz Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build RackViz — a rack equipment management desktop app with visual rack grid, drag-drop, network topology, port/cable management, reports, SSH terminal, and Tauri desktop packaging.

**Architecture:** React 18 + TypeScript + Vite frontend with Konva for 2D canvas and TanStack Query for state. Python FastAPI backend with SQLAlchemy ORM, Alembic migrations, and APScheduler for background tasks. Tauri 2 wraps everything as a cross-platform desktop app.

**Tech Stack:** React 18, TypeScript, Vite, TanStack Query v5, React-Konva, Ant Design 5, dagre, xterm.js, FastAPI, SQLAlchemy 2, Alembic, APScheduler, openpyxl, python-docx, weasyprint, asyncssh, Tauri 2

**Spec:** `docs/superpowers/specs/rackviz-design.md`

- [ ] **Step 1: Write `backend/app/services/icmp.py`**

```python
import subprocess
import re
import platform
import ipaddress
from typing import Optional
from concurrent.futures import ThreadPoolExecutor, as_completed
from ..core.config import settings

def ping(ip: str) -> Optional[float]:
    try:
        ipaddress.ip_address(ip)
    except ValueError:
        return None
    param = "-n" if platform.system().lower() == "windows" else "-c"
    timeout_param = "-w" if platform.system().lower() == "windows" else "-W"
    count = str(settings.icmp_count)
    timeout = str(settings.icmp_timeout)
    try:
        result = subprocess.run(
            ["ping", param, count, timeout_param, timeout, ip],
            capture_output=True, text=True, timeout=settings.icmp_timeout + 2,
        )
        if result.returncode != 0:
            return None
        match = re.search(r"(?:time|时间)[=<](\d+\.?\d*)\s*ms", result.stdout)
        if match:
            return float(match.group(1))
        return 0.0
    except (subprocess.TimeoutExpired, Exception):
        return None

def ping_many(ips: list[str], max_workers: int = 10) -> dict[str, Optional[float]]:
    results = {}
    with ThreadPoolExecutor(max_workers=max_workers) as ex:
        futures = {ex.submit(ping, ip): ip for ip in ips}
        for f in as_completed(futures):
            results[futures[f]] = f.result()
    return results
```

class PlacementConflictError(Exception):
    def __init__(self, conflicting_device_id: int, conflicting_device_name: str):
        self.conflicting_device_id = conflicting_device_id
        self.conflicting_device_name = conflicting_device_name
        super().__init__(f"U-slot conflict with device: {conflicting_device_name}")

def u_ranges_overlap(a_start: int, a_end: int, b_start: int, b_end: int) -> bool:
    return a_start <= b_end and b_start <= a_end

def validate_device_placement(db: Session, rack_id: int, start_u: int, end_u: int, exclude_device_id: int = None):
    q = db.query(Device).filter(Device.rack_id == rack_id)
    if exclude_device_id:
        q = q.filter(Device.id != exclude_device_id)
    for d in q:
        if d.start_u is None or d.end_u is None:
            continue
        if u_ranges_overlap(start_u, end_u, d.start_u, d.end_u):
            raise PlacementConflictError(d.id, d.name)

from sqlalchemy.orm import Session
from ..models.audit_log import AuditLog

def record_audit(db: Session, action: str, entity_type: str, entity_id: int,
                 payload_before: dict = None, payload_after: dict = None):
    log = AuditLog(
        action=action, entity_type=entity_type, entity_id=entity_id,
        payload_before=payload_before, payload_after=payload_after,
    )
    db.add(log)
    db.commit()

def undo(db: Session) -> dict | None:
    log = db.query(AuditLog).filter(
        AuditLog.undone == False
    ).order_by(AuditLog.id.desc()).first()
    if not log:
        return None
    entity = _get_entity(db, log.entity_type, log.entity_id)
    if entity and log.payload_before:
        for k, v in log.payload_before.items():
            if hasattr(entity, k):
                setattr(entity, k, v)
        db.commit()
    log.undone = True
    db.commit()
    return {"undone": log.action, "entity_type": log.entity_type, "entity_id": log.entity_id}

def redo(db: Session) -> dict | None:
    log = db.query(AuditLog).filter(
        AuditLog.undone == True
    ).order_by(AuditLog.id.asc()).first()
    if not log:
        return None
    entity = _get_entity(db, log.entity_type, log.entity_id)
    if entity and log.payload_after:
        for k, v in log.payload_after.items():
            if hasattr(entity, k):
                setattr(entity, k, v)
        db.commit()
    log.undone = False
    db.commit()
    return {"redone": log.action, "entity_type": log.entity_type, "entity_id": log.entity_id}

def _get_entity(db: Session, entity_type: str, entity_id: int):
    from ..models.device import Device
    from ..models.rack import Rack
    from ..models.device_model import DeviceModel
    from ..models.connection import Connection
    model_map = {
        "device": Device, "rack": Rack, "device_model": DeviceModel,
        "connection": Connection,
    }
    model = model_map.get(entity_type)
    if not model:
        return None
    return db.query(model).filter(model.id == entity_id).first()
