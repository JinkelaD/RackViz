from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from typing import List
from ..core.database import get_db
from ..models.device_model import DeviceModel
from ..schemas.device_model import (
    DeviceModelCreate, DeviceModelUpdate, DeviceModelResponse
)

router = APIRouter(prefix="/device-models", tags=["device-models"])

@router.get("/", response_model=List[DeviceModelResponse])
def list_models(db: Session = Depends(get_db)):
    return db.query(DeviceModel).all()

@router.post("/", response_model=DeviceModelResponse, status_code=201)
def create_model(data: DeviceModelCreate, db: Session = Depends(get_db)):
    model = DeviceModel(**data.model_dump())
    db.add(model)
    db.commit()
    db.refresh(model)
    return model

@router.get("/{model_id}", response_model=DeviceModelResponse)
def get_model(model_id: int, db: Session = Depends(get_db)):
    model = db.query(DeviceModel).filter(DeviceModel.id == model_id).first()
    if not model:
        raise HTTPException(status_code=404, detail="Device model not found")
    return model

@router.put("/{model_id}", response_model=DeviceModelResponse)
def update_model(model_id: int, data: DeviceModelUpdate, db: Session = Depends(get_db)):
    model = db.query(DeviceModel).filter(DeviceModel.id == model_id).first()
    if not model:
        raise HTTPException(status_code=404, detail="Device model not found")
    for key, val in data.model_dump(exclude_unset=True).items():
        setattr(model, key, val)
    db.commit()
    db.refresh(model)
    return model

@router.delete("/{model_id}", status_code=204)
def delete_model(model_id: int, db: Session = Depends(get_db)):
    model = db.query(DeviceModel).filter(DeviceModel.id == model_id).first()
    if not model:
        raise HTTPException(status_code=404, detail="Device model not found")
    db.delete(model)
    db.commit()

- [ ] **Step 1: Write `backend/app/services/icmp.py`**

```python
import subprocess
import re
import platform
from typing import Optional
from ..core.config import settings

def ping(ip: str) -> Optional[float]:
    """Ping an IP address, return latency in ms or None if failed."""
    param = "-n" if platform.system().lower() == "windows" else "-c"
    timeout_param = "-w" if platform.system().lower() == "windows" else "-W"
    count = str(settings.icmp_count)
    timeout = str(settings.icmp_timeout)

    try:
        result = subprocess.run(
            ["ping", param, count, timeout_param, timeout, ip],
            capture_output=True, text=True, timeout=settings.icmp_timeout + 2,
        )
        if result.returncode != 0:
            return None
        # Parse latency from output: "time=1.23ms" or "时间=1.23ms"
        match = re.search(r"time[=<](\d+\.?\d*)\s*ms", result.stdout)
        if match:
            return float(match.group(1))
        return 0.0
    except (subprocess.TimeoutExpired, Exception):

from io import BytesIO
from openpyxl import load_workbook
from sqlalchemy.orm import Session

def import_devices_excel(file_bytes: bytes, db: Session) -> dict:
    wb = load_workbook(filename=BytesIO(file_bytes))
    ws = wb.active
    rows = list(ws.iter_rows(values_only=True))

    imported = 0
    errors = []

    for row_idx, row in enumerate(rows):
        if row_idx < 2:  # Skip header rows
            continue
        if not row or not row[0]:
            continue
        try:
            name = str(row[0]).strip()
            model_name = str(row[1]).strip() if len(row) > 1 and row[1] else ""
            device_type = str(row[2]).strip() if len(row) > 2 and row[2] else "server"
            location = str(row[3]).strip() if len(row) > 3 and row[3] else ""
            manufacturer = str(row[4]).strip() if len(row) > 4 and row[4] else ""
            function = str(row[5]).strip() if len(row) > 5 and row[5] else ""

            # Parse location: "机柜01，34-35U"
            rack_name = ""
            start_u = None
            end_u = None
            if location:
                parts = location.replace("，", ",").split(",")
                if len(parts) >= 1:
                    rack_name = parts[0].strip()
                if len(parts) >= 2:
                    u_part = parts[1].strip().replace("U", "").replace("u", "")
                    if "-" in u_part:
                        u_parts = u_part.split("-")
                        try:
                            start_u = int(u_parts[0])
                            end_u = int(u_parts[1])
                        except ValueError:
                            pass

            # Find or create rack
            rack = None
            if rack_name:
                rack = db.query(Rack).filter(Rack.name == rack_name).first()
                if not rack:
                    rack = Rack(name=rack_name, height_u=42)
                    db.add(rack)
                    db.flush()

            # Find or create device model
            model = None
            if model_name:
                model = db.query(DeviceModel).filter(DeviceModel.name == model_name).first()
                if not model:
                    model = DeviceModel(
                        name=model_name, type=device_type,
                        manufacturer=manufacturer, height_u=1,
                    )
                    db.add(model)
                    db.flush()

            device = Device(
                name=name,
                device_model_id=model.id if model else None,
                rack_id=rack.id if rack else None,
                start_u=start_u,
                end_u=end_u,
                function=function,
            )
            db.add(device)
            imported += 1
        except Exception as e:
            errors.append(f"Row {row_idx + 1}: {str(e)}")

    db.commit()
    return {"imported": imported, "errors": errors}
```

  const res = await fetch(`${BASE}${url}`, {

  open: boolean;
  device: Device | null;
  racks: Rack[];
  onClose: () => void;
  onSave: (id: number, data: Partial<Device>) => Promise<void>;
}

export default function EditDeviceModal({ open, device, racks, onClose, onSave }: Props) {
  const [loading, setLoading] = useState(false);
  const [form] = Form.useForm();

  useEffect(() => {
    if (device && open) {
      form.setFieldsValue({
        ...device,

import { useState, useMemo, useCallback } from 'react';
import { useOutletContext } from 'react-router-dom';
import { message } from 'antd';
import { useRacks } from '../hooks/useRacks';
import { useDevices } from '../hooks/useDevices';
import { useDeviceModels } from '../hooks/useDeviceModels';
import { EditMode, ViewMode, Device, DeviceType, DeviceModel } from '../types';
import RackGrid from '../components/RackGrid';
import DeviceLibrary from '../components/DeviceLibrary';
import DeviceDetail from '../components/DeviceDetail';
import EditDeviceModal from '../components/EditDeviceModal';

export default function RackView() {
  const { mode, view, zoom, searchQuery } = useOutletContext<{
    mode: EditMode; view: ViewMode; zoom: number; searchQuery: string;
  }>();

  const { racks } = useRacks();
  const { devices, create, update, remove, refreshStatus } = useDevices();
  const { models } = useDeviceModels();
  const [selectedDeviceId, setSelectedDeviceId] = useState<number | null>(null);
  const [editDevice, setEditDevice] = useState<Device | null>(null);

  const modelTypes: Record<number, DeviceType> = useMemo(() => {
    const map: Record<number, DeviceType> = {};
    models.forEach(m => { map[m.id] = m.type; });
    return map;
  }, [models]);

  const maxCol = Math.max(...racks.map(r => r.col), 0) + 1;

  const handleSelectDevice = useCallback((id: number | null) => {
    if (mode === 'delete' && id !== null) {
      remove(id);
      message.success('设备已删除');
      setSelectedDeviceId(null);
    } else if (mode === 'edit' && id !== null) {
      const device = devices.find(d => d.id === id);
      if (device) setEditDevice(device);
    } else {
      setSelectedDeviceId(id);
    }
  }, [mode, remove, devices]);

  const handleDoubleClickDevice = useCallback((device: Device) => {
    setEditDevice(device);
  }, []);

  const handleDeviceMove = useCallback(async (deviceId: number, du: number) => {
    const device = devices.find(d => d.id === deviceId);
    if (!device || device.start_u == null || device.end_u == null) return;
    const rack = racks.find(r => r.id === device.rack_id);
    if (!rack) return;
    const newStart = device.start_u + du;
    const newEnd = device.end_u + du;
    if (newStart < 1 || newEnd > rack.height_u) return;
    await update(deviceId, { start_u: newStart, end_u: newEnd });
  }, [devices, racks, update]);

  const handleDropOnRack = useCallback(async (rackId: number, uPosition: number, modelId: number) => {
    const model = models.find(m => m.id === modelId);
    if (!model) return;
    const endU = uPosition + model.height_u - 1;
    await create({
      name: model.name,
      device_model_id: modelId,
      rack_id: rackId,
      start_u: uPosition,
      end_u: endU,
      power_watt: model.power_watt,
    });
    message.success(`已放置: ${model.name}`);
  }, [models, create]);

  const handleEditDeviceSave = useCallback(async (id: number, data: Partial<Device>) => {
    await update(id, data);
    message.success('设备已更新');
  }, [update]);

  const selectedDevice = devices.find(d => d.id === selectedDeviceId) || null;

  return (
    <div className="rack-view">
      <div className="canvas-area">
        <RackGrid
          racks={racks}
          devices={devices}
          modelTypes={modelTypes}
          selectedDeviceId={selectedDeviceId}
          onSelectDevice={handleSelectDevice}
          onDoubleClickDevice={handleDoubleClickDevice}
          onDeviceMove={handleDeviceMove}
          onDropOnRack={handleDropOnRack}
          zoom={zoom}
          cols={maxCol}
        />
      </div>
      <aside className="sidebar">
        <DeviceLibrary models={models} />
        <DeviceDetail
          device={selectedDevice}
          modelType={selectedDevice ? modelTypes[selectedDevice.device_model_id ?? 0] : undefined}
        />
      </aside>
      <EditDeviceModal
        open={editDevice !== null}
        device={editDevice}
        racks={racks}
        onClose={() => setEditDevice(null)}
        onSave={handleEditDeviceSave}
      />
    </div>
  );
}
